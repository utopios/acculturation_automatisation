📋 Pipeline GitLab CI

stages:
  - lint
  - test
  - build
  - deploy-dev
  - deploy-staging
  - deploy-prod

lint:
  stage: lint
  script:
    - flake8 app/ --max-line-length=120
    - ansible-lint playbooks/*.yml

unit_tests:
  stage: test
  services:
    - postgres:15-alpine
  script:
    - pytest tests/ -v --cov=.

build:
  stage: build
  script:
    - docker build -t $IMAGE:$TAG .
    - docker push $IMAGE:$TAG

deploy_prod:
  stage: deploy-prod
  when: manual
  environment:
    name: production

🔄 Pipeline Drone CI

kind: pipeline
type: docker
name: telco-provisioning

steps:
  - name: lint
    image: python:3.11-slim
    commands:
      - pip install flake8
      - flake8 app/

  - name: test
    image: python:3.11-slim
    commands:
      - pip install pytest
      - pytest tests/ -v

  - name: build
    image: docker:24
    commands:
      - docker build -t app .

  - name: deploy
    image: docker:24
    commands:
      - docker push registry:5000/app
    when:
      branch: main

📋 Commandes de la Démo

docker-compose up -d Démarrer la stack CI/CD complète
curl http://localhost:3000 Accéder à Gitea (Git server)
curl http://localhost:8080 Accéder à Drone CI
curl http://localhost:5001/health Health check environnement Dev
curl -X POST http://localhost:5001/api/provision -H "Content-Type: application/json" -d '{"service_type":"voip","customer_id":"C001"}' Tester le provisioning API
docker logs -f cicd-drone-runner Suivre les logs du runner CI
curl http://localhost:5000/v2/_catalog Lister les images du registry
